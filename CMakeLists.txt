cmake_minimum_required(VERSION 3.5...3.29)

project(duckdb-odbc C CXX)

find_package(ODBC)
find_package(Threads REQUIRED)

set(CMAKE_CXX_STANDARD "17" CACHE STRING "C++ standard to enforce")
set(CMAKE_VERBOSE_MAKEFILE OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT ODBC_FOUND)
  message(FATAL_ERROR "No ODBC found; you need to install unixodbc-dev.")
endif()

if(OSX_BUILD_AMD64)
  set(CMAKE_OSX_ARCHITECTURES "x86_64" CACHE STRING "Build architectures for Mac OS X" FORCE)
endif()

# The DuckDB engine. The driver only uses DuckDB through its stable C++ API, vendored in third_party/duckdb_cpp, and
# statically links an engine built from the same revision. scripts/build_duckdb.py builds one; `make` runs it when
# DUCKDB_ROOT is not given.

include(${CMAKE_CURRENT_LIST_DIR}/third_party/duckdb_cpp/duckdb_revision.cmake)

set(DUCKDB_ROOT "" CACHE PATH "Install prefix of the DuckDB engine to link (see scripts/build_duckdb.py)")
if(NOT DUCKDB_ROOT)
  message(FATAL_ERROR "DUCKDB_ROOT is not set. Build the DuckDB engine with "
                      "'python3 scripts/build_duckdb.py --install-dir <dir>' and pass -DDUCKDB_ROOT=<dir>, or use "
                      "'make', which does both.")
endif()

set(DuckDB_USE_STATIC_LIBS ON)
find_package(DuckDB CONFIG REQUIRED PATHS "${DUCKDB_ROOT}" NO_DEFAULT_PATH)
foreach(REQUIRED_EXTENSION core_functions parquet icu json)
  if(NOT "${REQUIRED_EXTENSION}" IN_LIST DuckDB_EXTENSIONS)
    message(FATAL_ERROR "The DuckDB engine in ${DUCKDB_ROOT} does not include the ${REQUIRED_EXTENSION} extension "
                        "(it has: ${DuckDB_EXTENSIONS}); build it with scripts/build_duckdb.py.")
  endif()
endforeach()
message(STATUS "DuckDB: ${DUCKDB_ROOT} (${DUCKDB_GIT_VERSION}, ${DUCKDB_GIT_SHA})")

# The C++ API is compiled into the driver; only the driver's SQL* entry points are exported.
add_library(odbc_duckdb_cpp_api OBJECT third_party/duckdb_cpp/duckdb_cpp.cpp)
target_include_directories(odbc_duckdb_cpp_api PUBLIC ${CMAKE_CURRENT_LIST_DIR}/third_party/duckdb_cpp)
set_target_properties(odbc_duckdb_cpp_api PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
  target_compile_options(odbc_duckdb_cpp_api PRIVATE /bigobj /utf-8)
endif()

# Registers the statically linked extensions, see duckdb_register_static_extensions. Generated by build_duckdb.py.
set(DUCKDB_STATIC_EXTENSION_LOADER "${DUCKDB_ROOT}/share/duckdb_odbc/static_extension_loader.c")
if(NOT EXISTS "${DUCKDB_STATIC_EXTENSION_LOADER}")
  message(FATAL_ERROR "${DUCKDB_STATIC_EXTENSION_LOADER} not found; build the engine with scripts/build_duckdb.py.")
endif()
add_library(odbc_duckdb_extension_loader OBJECT "${DUCKDB_STATIC_EXTENSION_LOADER}")
# only for duckdb_static_extension.h: the driver itself must not see DuckDB's internal headers
target_include_directories(odbc_duckdb_extension_loader PRIVATE "${DUCKDB_ROOT}/include")
set_target_properties(odbc_duckdb_extension_loader PROPERTIES C_VISIBILITY_PRESET hidden)

include_directories(${CMAKE_CURRENT_LIST_DIR}/third_party/duckdb_cpp ${ODBC_INCLUDE_DIRS} include third_party)

set(LINK_LIB_LIST "")

if(ODBC_FOUND)
  message(STATUS "ODBC library found at: " ${ODBC_LIBRARIES})
  get_filename_component(ODBC_LIB_DIR ${ODBC_LIBRARIES} DIRECTORY)
  message(STATUS "ODBC library path: " ${ODBC_LIB_DIR})
  find_library(ODBCINST_LIB NAMES odbcinst libodbcinst HINTS ${ODBC_LIB_DIR})
  if(ODBCINST_LIB)
    list(APPEND LINK_LIB_LIST ${ODBCINST_LIB})
    add_definitions(-DODBC_LINK_ODBCINST)
    message(STATUS "Found ODBCINST_LIB: " ${ODBCINST_LIB})
  endif()

  if(WIN32)
    string(REGEX MATCH "^v([0-9]+)\\.([0-9]+)\\.([0-9]+)" _ "${DUCKDB_GIT_VERSION}")
    set(DUCKDB_ENGINE_VERSION ${CMAKE_MATCH_1}${CMAKE_MATCH_2}${CMAKE_MATCH_3})
    message(STATUS "Version number parsed out from the vendored DuckDB revision: ${DUCKDB_ENGINE_VERSION}")
    set(DUCKDB_ODBC_VERSION "00")
    set(DUCKDB_WIN_COMMA_VERSiON "${DUCKDB_ENGINE_VERSION},${DUCKDB_ODBC_VERSION},00,00")
    set(DUCKDB_WIN_DOT_VERSiON "${DUCKDB_ENGINE_VERSION}.${DUCKDB_ODBC_VERSION}.00.00")
    configure_file(${CMAKE_CURRENT_LIST_DIR}/duckdb_odbc.rc.in ${CMAKE_CURRENT_BINARY_DIR}/duckdb_odbc.rc)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/bin)
    add_subdirectory(winsetup)
    list(APPEND LINK_LIB_LIST $<$<PLATFORM_ID:Windows>:odbccp32>)
    list(APPEND LINK_LIB_LIST
         $<$<PLATFORM_ID:Windows>:legacy_stdio_definitions>)
  endif()
endif()

if (CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_definitions(-DODBC_DEBUG)
endif()

add_subdirectory(src)

set(ALL_OBJECT_FILES ${ALL_OBJECT_FILES} $<TARGET_OBJECTS:odbc_duckdb_cpp_api>
                     $<TARGET_OBJECTS:odbc_duckdb_extension_loader>)
if(WIN32)
  list(APPEND ALL_OBJECT_FILES
    duckdb_odbc.def
    ${CMAKE_CURRENT_BINARY_DIR}/duckdb_odbc.rc)
endif()

add_library(duckdb_odbc SHARED ${ALL_OBJECT_FILES})
if(WIN32)
  target_compile_options(duckdb_odbc PRIVATE /bigobj /utf-8)
endif()

set_target_properties(duckdb_odbc PROPERTIES DEFINE_SYMBOL "DUCKDB_ODBC_API")
target_link_libraries(duckdb_odbc ${LINK_LIB_LIST})
# the extension archives go before the engine, which they depend on
target_link_libraries(
  duckdb_odbc
  core_functions_extension
  parquet_extension
  icu_extension
  json_extension
  duckdb_loadable_extensions
  duckdb_httplib
  duckdb_static
  Threads::Threads
  ${CMAKE_DL_LIBS})
if(NOT WIN32)
  if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(duckdb_odbc PRIVATE
      -fvisibility=hidden
      -fvisibility-inlines-hidden)
    target_link_options(duckdb_odbc PRIVATE
      -Wl,-Bsymbolic
      -Wl,--exclude-libs,ALL
      -Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/duckdb_odbc.map)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
    target_compile_options(duckdb_odbc PRIVATE
      -fvisibility=hidden
      -fvisibility-inlines-hidden)
    target_link_options(duckdb_odbc PRIVATE
      -Wl,-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/duckdb_odbc.exp)
  endif()
endif()

if(NOT CLANG_TIDY)
  add_subdirectory(test)
endif()
